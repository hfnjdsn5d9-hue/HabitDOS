  const ds=week(d), groups={};

function monthDays(m){

  let n=new Date(m.getFullYear(),m.getMonth()+1,0).getDate();

  return Array.from({length:n},(_,i)=>new Date(m.getFullYear(),m.getMonth(),i+1));

}

function monthProg(h,m){

  let ds=monthDays(m),f=h.freq;

  if(f.type==="times"){

    let first=new Date(m.getFullYear(),m.getMonth(),1),last=new Date(m.getFullYear(),m.getMonth()+1,0),fm=mon(first),lm=mon(last),target=0,doneN=0;

    for(let w=new Date(fm);w<=lm;w=add(w,7)){

      if(w.getMonth()!=m.getMonth()||w.getFullYear()!=m.getFullYear())continue;

      let n=week(w).filter(d=>done(h,d)).length;

      target+=f.count;doneN+=Math.min(f.count,n);

    }

    return{done:doneN,target,p:target?Math.round(doneN/target*100):0};

  }

  let dueDays=f.type==="daily"?ds:ds.filter(d=>f.days.includes(d.getDay()+1)),dn=dueDays.filter(d=>done(h,d)).length;

  return{done:dn,target:dueDays.length,p:dueDays.length?Math.round(dn/dueDays.length*100):0};

}

function weekNumber(d){

  const x=new Date(Date.UTC(d.getFullYear(),d.getMonth(),d.getDate()));

  x.setUTCDate(x.getUTCDate()+4-(x.getUTCDay()||7));

  const y=new Date(Date.UTC(x.getUTCFullYear(),0,1));

  return Math.ceil((((x-y)/86400000)+1)/7);

}


function historyWeeks(y){

  let out=[];

  const start=new Date(y,0,1), end=new Date(y,11,31), w=mon(start);

  for(let d=new Date(w);d<=end;d=add(d,7)){

    if(d<today() && d.getFullYear()===y && d.getMonth()>=0) out.push(d);

  }

  return out.reverse();

}

function restore(e){

  let file=e.target.files[0];if(!file)return;

  let r=new FileReader();

  r.onload=()=>{

    try{

      let x=JSON.parse(r.result);

      if(!Array.isArray(x.habits)||!Array.isArray(x.records))throw 0;

      habits=x.habits;records=x.records;todos=Array.isArray(x.todos)?x.todos:[];

      save();render();alert("HabitDOS-Daten wurden importiert.");

    }catch(err){alert("Ungültige HabitDOS-Backup-Datei.");}

  };

  r.readAsText(file);

}


/* ── Swipe Navigation ── */

let touchStartX=0,touchStartY=0,touchStartTime=0;

const SWIPE_THRESHOLD=60;  // min horizontal px

const SWIPE_MAX_Y=80;      // max vertical drift

const SWIPE_MAX_TIME=400;  // max duration ms


document.addEventListener("touchstart",e=>{

  if(e.touches.length!==1)return;

  touchStartX=e.touches[0].clientX;

  touchStartY=e.touches[0].clientY;

  touchStartTime=Date.now();

},{passive:true});


document.addEventListener("touchend",e=>{

  if(!touchStartX && !touchStartY)return;

  const dx=e.changedTouches[0].clientX-touchStartX;

  const dy=e.changedTouches[0].clientY-touchStartY;

  const dt=Date.now()-touchStartTime;

  touchStartX=touchStartY=0;

  if(dt>SWIPE_MAX_TIME)return;

  if(Math.abs(dy)>SWIPE_MAX_Y)return;

  if(Math.abs(dx)<SWIPE_THRESHOLD)return;

  // Determine direction

  const idx=TAB_ORDER.indexOf(tab);

  if(idx<0)return;

  if(dx<0 && idx<TAB_ORDER.length-1){

    // swipe left → next

    animateSwipe("left",()=>setTab(TAB_ORDER[idx+1]));

  }else if(dx>0 && idx>0){

    // swipe right → prev

    animateSwipe("right",()=>setTab(TAB_ORDER[idx-1]));

  }

},{passive:true});


function animateSwipe(dir,cb){

  const main=document.querySelector("main");

  if(!main){cb();return;}

  main.classList.add(dir==="left"?"swipe-left":"swipe-right");

  setTimeout(()=>{

    cb();

    requestAnimationFrame(()=>{

      const m=document.querySelector("main");

      if(m){m.classList.remove("swipe-left","swipe-right");}

    });

  },120);

}

}

let editId=null,editFreq={type:"daily"},editDays=[];


function editor(id){

  if(editId){

    if(!h)return;

function setFreqType(t){

  editFreq={type:t};

  if(t==="times")editFreq.count=3;

  if(t==="days")editDays=[];

  renderEditor();

}

function changeCount(n){

  editFreq.count=Math.max(1,Math.min(7,(editFreq.count||3)+n));

  renderEditor();

}

function toggleDay(d){

  if(editDays.includes(d))editDays=editDays.filter(x=>x!==d);

  else editDays.push(d);

  editDays.sort((a,b)=>a-b);

  renderEditor();

}

function cancelEditor(){render();}

function saveHabit(){

  let name=$("#ename").value.trim();

  if(!name){alert("Name erforderlich.");return;}

  let freq=cloneFreq(editFreq);

  if(freq.type==="days")freq.days=[...editDays];

  if(editId){

    let h=habits.find(x=>x.id===editId);

    if(h){

      let oldFreq=JSON.stringify(h.freq);

      let newFreq=JSON.stringify(freq);

      if(oldFreq!==newFreq){

        if(!h.versions)h.versions=[{from:"2000-01-01",freq:cloneFreq(h.freq)}];

        h.versions.push({from:k(today()),freq:cloneFreq(freq)});

      }

      h.name=name;h.freq=freq;

    }

  }else{

    habits.push({id:uid(),name,freq,arch:false,versions:[{from:"2000-01-01",freq:cloneFreq(freq)}]});

  }

  save();render();

}

function deleteHabit(){

  if(!confirm("Habit wirklich löschen? Alle Daten gehen verloren."))return;

  habits=habits.filter(x=>x.id!==editId);

  records=records.filter(r=>r.h!==editId);

  save();render();

}

function archiveHabit(){

  let h=habits.find(x=>x.id===editId);

  if(h){h.arch=true;save();render();}

}

function unarchiveHabit(){

  let h=habits.find(x=>x.id===editId);

  if(h){h.arch=false;save();render();}

}
